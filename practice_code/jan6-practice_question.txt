class DoublyLinkedList{
private:
   Node* head_;
   Node* tail_;
   Node* sentinel_;
   class Node{
   public:
      int data;
      Node* next = nullptr;
      Node* prev = nullptr;
   };

public:
   DoublyLinkedList(): head_(nullptr), tail_(nullptr), sentinel_(new Node()){
      sentinel_->next = sentinel_;
      sentinel_->prev = sentinel_;
   }

   Iterator begin() {return Iterator(head_);}
   Iterator end() {return Iterator(sentinel_);}

   class Iterator: public std::bidirectional-something{
   private:
      Node* pos_;
   public:
      Iterator():pos_(nullptr){}
      Iterator(Node* pos): pos_(pos){}
      Iterator& operator--(){
         if (pos_){
            pos_ = pos_->prev;
            return *this;
         }
      }
   };
};
