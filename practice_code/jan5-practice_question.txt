class Buffer {
public:
   Buffer(size_t n):  n_(n), data_(new int[n_]){
   }

   ~Buffer(){
      delete[] data_;
   }

   Buffer(const Buffer& other): n_(n), data_(new int[n_]){
      for (size_t i = 0; i < n_; ++i){
         data_[i] = other.data_[i];
      }
   }

   Buffer& operator=(const Buffer& rhs){
      if (this == &rhs) return *this;
      delete[] data_;
      n_ = other.n_;
      data_ = new int[n_];
      for (size_t i = 0; i < n_; ++i){
         data_[i] = other.data_[i];
      }
      return *this;
   }

   size_t size() const{return n_;}
   int& operator[](size_t i){
      if (i < 0 || i >= n_) throw;
      return data_[i];
   }

   const int& operator[](size_t i) const{
      if (i < 0 || i >= n_) throw;
      return data_[i];
   }

private:
   size_t n_;
   int* data_;
};
