void push_front(Node*& head, Node*& tail, const T& val){
   Node* temp = new Node(val);
   if (!head && !tail) {
      head = tail = temp;
      return;
   }
   temp->next = head;
   head->prev = temp;
   head = temp;
}

void push_back(Node*& head, Node*& tail, const T& val){
   Node* temp = new Node(val);
   if (!head && !tail) {
      head = tail = temp;
      return;
   }
   tail->next = temp;
   temp->prev = tail;
   tail = temp;
}

bool pop_front(Node*& head, Node*& tail){
   if (!head && !tail) return false;
   if (head == tail) {
      delete head;
      head = tail = nullptr;
      return true;
   }
   Node* temp = head;
   head = head->next;
   head->prev = nullptr;
   delete temp;
   temp = nullptr;
   return true;
}

bool pop_back(Node*& head, Node*& tail){
   if (!head && !tail) return false;
   if (head == tail) {
      delete tail;
      head = tail = nullptr;
      return true;
   }
   Node* temp = tail;
   tail = tail->prev;
   tail->next = nullptr;
   delete temp;
   temp = nullptr;
   return true;
}

bool checkLinks(Node* head, Node* tail){
   if (!head) return tail==nullptr;
   if (!tail) return head==nullptr;
   if (head->prev != nullptr || tail->next!= nullptr) return false;

   int count = 0;
   Node* curr = head;
   while (curr){
      if (curr->next && curr->next->prev != curr) return false;
      curr = curr->next;
      count++;
   }
   int count2 = 0;
   Node* curr2 = tail;
   while (curr2){
      curr2 = curr2->prev;
      count2++;
   }

   return count2==count;
}
